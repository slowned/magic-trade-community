import requests

def search_card(card_name):
    url = f"https://api.scryfall.com/cards/named?fuzzy={card_name}"
    response = requests.get(url)

    if response.status_code == 200:
        card_data = response.json()
        return {
            "name": card_data.get('name'),
            "set": card_data.get('set_name'),
            "rarity": card_data.get('rarity'),
            "image_url": card_data['image_uris']['normal'] if 'image_uris' in card_data else 'No image available',
            "oracle_text": card_data.get('oracle_text', 'No text available')
        }
    else:
        return f"Card '{card_name}' not found."

def search_cards(cards):
    for card in cards:
        card_info = search_card(card)
        print(f"Name: {card_info['name']}")
        print(f"Set: {card_info['set']}")
        print(f"Rarity: {card_info['rarity']}")
        print(f"Image: {card_info['image_url']}")
        print(f"Text: {card_info['oracle_text']}")
        print("-" * 40)
